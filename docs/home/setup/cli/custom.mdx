A custom CLI gives an agent your own command: a client for your service, a
tool over workspace files, or a script. Full guide:
[Python](/python/cli/custom) · [TypeScript](/typescript/cli/custom).

## Script CLI

A script reads its own arguments and needs no Mirage imports. Name it in the
workspace config, with a machine that runs its language:

```yaml workspace.yaml
mounts:
  /:
    vfs: ram
machines:
  monty:
    provider: in_process
    runtime: monty
clis:
  hello:
    script: ./hello.py
    machine: in_process
    config:
      greeting: hi
```

```python hello.py
import json
import os

config = json.loads(os.environ["MIRAGE_CLI_CONFIG"])
print(config["greeting"], *argv[1:])
```

`hello sam` prints `hi sam`: the words after `hello` arrive as `argv`, and the
entry's `config` as JSON in `MIRAGE_CLI_CONFIG`.

## Typed CLI

A typed CLI declares its subcommands and flags, so Mirage parses each line,
refuses a bad one and generates `--help`. Register it on an existing `ws`:

<Tabs>
  <Tab title="Python" icon="/images/python-logo.svg">
    ```python
    from mirage import Argument, CLI, CLIHandler, CommandSpec, Context, FlagView, PathSpec
    from mirage.accessor import NOOPAccessor
    from mirage.io import IOResult

    async def greet(accessor: NOOPAccessor, paths: list[PathSpec],
                    texts: list[str], ctx: Context) -> tuple[bytes, IOResult]:
        name = FlagView(ctx.flags, ctx.spec).as_str("name")
        return f"hi {name}\n".encode(), IOResult()

    HELLO = CLI(
        spec=CommandSpec(name="hello", arguments=(Argument("--name", required=True),)),
        handlers={"": CLIHandler(fn=greet)},
    )
    ws.register_cli("hello", HELLO)
    ```
  </Tab>
  <Tab title="TypeScript" icon="/images/typescript-logo.svg">
    ```typescript
    import {
      Argument,
      CLI,
      CLIHandler,
      CommandSpec,
      type CommandFnResult,
      type Context,
      FlagView,
      IOResult,
      type PathSpec,
    } from '@struktoai/mirage-node'
    import type { NOOPAccessor } from '@struktoai/mirage-core/accessor/base'

    function greet(
      _accessor: NOOPAccessor,
      _paths: PathSpec[],
      _texts: string[],
      ctx: Context,
    ): CommandFnResult {
      const name = new FlagView(ctx.flags, ctx.spec).asStr('name')
      return [new TextEncoder().encode(`hi ${name}\n`), new IOResult()]
    }

    const HELLO = new CLI({
      spec: new CommandSpec({ name: 'hello', arguments: [new Argument('--name', { required: true })] }),
      handlers: { '': new CLIHandler({ fn: greet }) },
    })
    ws.registerCli('hello', HELLO)
    ```
  </Tab>
</Tabs>

`hello --name sam` prints `hi sam`. Without `--name`, Mirage refuses the line
and exits 2.
